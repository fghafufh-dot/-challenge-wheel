// منتصف خانة اللاعب العشوائي const playerAngle = randomIndex * slice + (slice / 2); // عدد اللفات const extraSpins = 360 * 6; // تدوير العجلة بحيث الخانة المختارة تقترب من السهم const targetRotation = extraSpins + (360 - playerAngle); showSelectedPlayer(); }, 4100); } // ===================================== // تحديد اللاعب الموجود تحت رأس السهم // ===================================== function getPlayerUnderPointer(){ const count = players.length;
